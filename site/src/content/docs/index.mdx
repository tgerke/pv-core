import { Card, CardGrid } from "@astrojs/starlight/components";

<CardGrid>
  <Card title="Cases shaped by E2B(R3)" icon="document">
    Reporters, patient, coded events with the six seriousness criteria, drugs, drug-by-event
    assessments, narrative. Versions are editable until the first signature and immutable
    after it; follow-ups and corrections are new versions.
  </Card>
  <Card title="Clocks derived, never maintained" icon="approve-check">
    Reporting rules are rows: destination, timeline, and a test over the case. Obligations
    materialize the moment a case is a valid ICSR; due soon, overdue, submitted,
    acknowledged, superseded, and waived are computed on every read.
  </Card>
  <Card title="Compliance as schema" icon="seti:lock">
    A hash-chained audit trail written by database triggers, immutable signatures and
    submissions, signatures bound to the version hash, re-authentication to sign, and a
    runtime role that can only run DML.
  </Card>
  <Card title="Made for the R seat too" icon="laptop">
    The web app consumes only the public API. The overdue queue, the DSUR line listing, and
    every derived view are one httr2 call or one read-only SQL connection away.
  </Card>
</CardGrid>
